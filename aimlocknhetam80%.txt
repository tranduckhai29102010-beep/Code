using System;

namespace FFCustomAim
{
    static class GamePackages
    {
        public const string PACKAGE1 = "com.dts.freefire";

    struct Vector3
    {
        public float x;
        public float y;
        public float z;

        public Vector3(float xv, float yv, float zv)
        {
            x = xv;
            y = yv;
            z = zv;
        }

        public override string ToString()
        {
            return "(" + x.ToString("F1") + ", " + y.ToString("F1") + ", " + z.ToString("F1") + ")";
        }
    }

    class SmoothAim60
    {
        private readonly int hexValue = 0x3C;

        public void Apply(ref Vector3 player, Vector3 target)
        {
            float alpha = 0.6f;
            float beta = 1.0f - alpha;

            player.x = player.x * beta + target.x * alpha;
            player.y = player.y * beta + target.y * alpha;
            player.z = player.z * beta + target.z * alpha;
        }
    }

    class Program
    {
        static void Main()
        {
            Vector3 player = new Vector3(0, 0, 0);
            Vector3 enemy = new Vector3(10, 5, 0);

            var smoother = new SmoothAim60();
            smoother.Apply(ref player, enemy);

            Console.WriteLine(player.ToString());
        }
    }
}
