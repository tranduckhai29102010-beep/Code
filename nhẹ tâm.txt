using UnityEngine;

public class SilkAim : MonoBehaviour
{
    public float sensitivity = 2.0f;
    public float emaAlpha   = 0.15f;
    public float tension    = 0.45f;
    public float deadzone   = 0.8f;
    public float kfQ        = 0.002f;
    public float kfR        = 0.04f;

    float   kfP=1,kfK,kfEx,kfEy,kfVx,kfVy;
    bool    kfInit;

    Vector2 evaPos,smoothPos,weightPos;
    bool    emaInit;

    float   noiseEst;
    float   subPx,subPy;
    Vector2 lastPos;
    bool    active;

    void KF(Vector2 z, out Vector2 r){
        if(!kfInit){kfEx=z.x;kfEy=z.y;kfInit=true;r=z;return;}
        float pxp=kfEx+kfVx,pyp=kfEy+kfVy,pp=kfP+kfQ;
        kfK=pp/(pp+kfR);
        float rx=pxp+kfK*(z.x-pxp),ry=pyp+kfK*(z.y-pyp);
        kfVx=rx-kfEx;kfVy=ry-kfEy;
        kfP=(1f-kfK)*pp;
        kfEx=rx;kfEy=ry;
        r=new Vector2(rx,ry);
    }

    public void Begin(Vector2 pos){
        lastPos=pos; smoothPos=pos; weightPos=pos;
        subPx=0; subPy=0; noiseEst=0;
        emaInit=false; active=true;
        kfInit=false; kfP=1; kfVx=0; kfVy=0;
    }

    public void End(){
        active=false; subPx=0; subPy=0;
    }

    public Vector2 Process(Vector2 raw){
        if(!active) return Vector2.zero;

        Vector2 kfPos;
        KF(raw,out kfPos);

        if(!emaInit){evaPos=kfPos;emaInit=true;}
        evaPos  =Vector2.Lerp(evaPos,   kfPos,  emaAlpha);
        weightPos=Vector2.Lerp(weightPos,evaPos, emaAlpha*0.5f);

        Vector2 blend=Vector2.Lerp(evaPos,weightPos,0.4f);
        Vector2 dst  =Vector2.Lerp(smoothPos,blend,tension);

        Vector2 delta=dst-lastPos;
        float   mag  =delta.magnitude;

        if(mag<deadzone){
            delta=Vector2.zero;
        } else {
            delta*=(mag-deadzone)/mag;
        }

        float m=delta.magnitude;
        noiseEst=noiseEst*0.95f+m*0.05f;
        float gate=Mathf.Clamp(noiseEst*0.9f,0.02f,deadzone*0.6f);
        if(m>0f&&m<gate) delta=Vector2.zero;

        delta*=sensitivity;

        subPx+=delta.x; subPy+=delta.y;
        float ox=Mathf.Floor(Mathf.Abs(subPx))*Mathf.Sign(subPx);
        float oy=Mathf.Floor(Mathf.Abs(subPy))*Mathf.Sign(subPy);
        subPx-=ox; subPy-=oy;

        smoothPos=dst;
        lastPos  =evaPos;

        return new Vector2(ox,oy);
    }

    public float Noise => noiseEst;
}
