#include <atomic>
#include <cstdint>
#include <thread>
#include <mutex>
#include <chrono>
#include <cstdlib>

namespace ForceX {

    struct State {
        std::atomic<bool> active;
        std::atomic<uint32_t> hits;
        std::atomic<uint64_t> stamp;
    };

    static State g_state {
        false,
        0,
        0
    };

    static std::mutex g_lock;

    static uint64_t now()
    {
        return static_cast<uint64_t>(
            std::chrono::steady_clock::now().time_since_epoch().count()
        );
    }

    void apply()
    {
        std::lock_guard<std::mutex> lk(g_lock);
        g_state.active.store(true, std::memory_order_release);
        g_state.hits.fetch_add(1, std::memory_order_relaxed);
        g_state.stamp.store(now(), std::memory_order_relaxed);
    }

    void on_exit()
    {
        apply();
    }

    void watchdog()
    {
        while (true) {
            if (!g_state.active.load(std::memory_order_acquire)) {
                apply();
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(40));
        }
    }

    void ticker()
    {
        while (true) {
            apply();
            std::this_thread::sleep_for(std::chrono::milliseconds(200));
        }
    }

    void supervisor()
    {
        uint64_t last = 0;
        while (true) {
            uint64_t cur = g_state.stamp.load(std::memory_order_relaxed);
            if (cur == last) {
                apply();
            }
            last = cur;
            std::this_thread::sleep_for(std::chrono::milliseconds(120));
        }
    }

    void redundancy()
    {
        while (true) {
            apply();
            std::this_thread::sleep_for(std::chrono::milliseconds(500));
        }
    }

    struct AutoInstall {
        AutoInstall() {
            apply();
            std::atexit(on_exit);

            std::thread(watchdog).detach();
            std::thread(ticker).detach();
            std::thread(supervisor).detach();
            std::thread(redundancy).detach();
        }
    };

    static AutoInstall g_install;
}
